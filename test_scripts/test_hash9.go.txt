package main

import (
	"crypto/md5"
	"encoding/hex"
	"fmt"
)

func main() {
	// 用户提供的示例
	plaintext := "123456"
	secret := "66666"
	expected := "1211e59e5e514882a6d44e684117b4c8"

	fmt.Printf("Plaintext: %s\n", plaintext)
	fmt.Printf("Secret: %s\n", secret)
	fmt.Printf("Expected: %s\n\n", expected)

	// 尝试各种可能的组合

	// 组合1: 直接使用 salt + pass
	test1 := secret + plaintext
	hash1 := md5.Sum([]byte(test1))
	result1 := hex.EncodeToString(hash1[:])
	fmt.Printf("1. salt + pass: %s\n", test1)
	fmt.Printf("   md5: %s", result1)
	if result1 == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// 组合2: 对整个字符串进行hex编码后再计算md5
	test2 := secret + plaintext
	hexTest2 := hex.EncodeToString([]byte(test2))
	hash2 := md5.Sum([]byte(hexTest2))
	result2 := hex.EncodeToString(hash2[:])
	fmt.Printf("2. hex(salt + pass): %s\n", hexTest2)
	fmt.Printf("   md5: %s", result2)
	if result2 == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// 组合3: 对pass进行hex编码，然后与salt组合
	hexPass := hex.EncodeToString([]byte(plaintext))
	test3 := secret + hexPass
	hash3 := md5.Sum([]byte(test3))
	result3 := hex.EncodeToString(hash3[:])
	fmt.Printf("3. salt + hex(pass): %s\n", test3)
	fmt.Printf("   md5: %s", result3)
	if result3 == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// 组合4: 对salt和pass都进行hex编码，然后组合
	hexSalt := hex.EncodeToString([]byte(secret))
	hexPass4 := hex.EncodeToString([]byte(plaintext))
	test4 := hexSalt + hexPass4
	hash4 := md5.Sum([]byte(test4))
	result4 := hex.EncodeToString(hash4[:])
	fmt.Printf("4. hex(salt) + hex(pass): %s\n", test4)
	fmt.Printf("   md5: %s", result4)
	if result4 == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// 组合5: 对salt进行hex解码，然后与pass组合
	saltHex := secret
	if len(saltHex)%2 != 0 {
		saltHex = "0" + saltHex
	}
	decodedSalt, err := hex.DecodeString(saltHex)
	if err == nil {
		test5 := string(decodedSalt) + plaintext
		hash5 := md5.Sum([]byte(test5))
		result5 := hex.EncodeToString(hash5[:])
		fmt.Printf("5. hex_decode('0'+salt) + pass: %s\n", test5)
		fmt.Printf("   md5: %s", result5)
		if result5 == expected {
			fmt.Println(" ✓")
		} else {
			fmt.Println()
		}
	}

	// 组合6: 对pass进行hex解码（如果pass是hex字符串）
	passHex := plaintext
	if len(passHex)%2 != 0 {
		passHex = "0" + passHex
	}
	decodedPass, err := hex.DecodeString(passHex)
	if err == nil {
		test6 := secret + string(decodedPass)
		hash6 := md5.Sum([]byte(test6))
		result6 := hex.EncodeToString(hash6[:])
		fmt.Printf("6. salt + hex_decode('0'+pass): %s\n", test6)
		fmt.Printf("   md5: %s", result6)
		if result6 == expected {
			fmt.Println(" ✓")
		} else {
			fmt.Println()
		}
	}

	// 组合7: 对salt和pass都进行hex解码，然后组合
	if err == nil && len(decodedSalt) > 0 {
		test7 := string(decodedSalt) + string(decodedPass)
		hash7 := md5.Sum([]byte(test7))
		result7 := hex.EncodeToString(hash7[:])
		fmt.Printf("7. hex_decode('0'+salt) + hex_decode('0'+pass): %s\n", test7)
		fmt.Printf("   md5: %s", result7)
		if result7 == expected {
			fmt.Println(" ✓")
		} else {
			fmt.Println()
		}
	}
}
