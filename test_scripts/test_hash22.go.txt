package main

import (
	"crypto/md5"
	"encoding/hex"
	"fmt"
)

func main() {
	plaintext := "123456"
	secret := "66666"

	// 方法：先对明文和盐分别进行MD5哈希，然后组合
	md5Pass := md5.Sum([]byte(plaintext))
	md5Salt := md5.Sum([]byte(secret))
	combined := hex.EncodeToString(md5Salt[:]) + hex.EncodeToString(md5Pass[:])
	hash := md5.Sum([]byte(combined))
	fmt.Printf("分别哈希后组合: %x\n", hash)

	// 预期结果
	fmt.Printf("预期结果: 1211e59e5e514882a6d44e684117b4c8\n")
}
