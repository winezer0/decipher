package main

import (
	"crypto/md5"
	"encoding/hex"
	"fmt"
)

func main() {
	// 用户提供的示例
	plaintext := "123456"
	secret := "66666"

	fmt.Printf("Plaintext: %s\n", plaintext)
	fmt.Printf("Secret: %s\n\n", secret)

	// 尝试直接对盐值进行 hex 解码
	decodedSalt, err := hex.DecodeString(secret)
	if err != nil {
		fmt.Printf("Error decoding salt: %v\n", err)
		return
	}

	fmt.Printf("Decoded salt: %s\n", string(decodedSalt))
	fmt.Printf("Decoded salt bytes: %v\n\n", decodedSalt)

	// 计算 md5($salt.$pass)
	hash1 := md5.Sum([]byte(string(decodedSalt) + plaintext))
	result1 := hex.EncodeToString(hash1[:])
	fmt.Printf("md5($salt.$pass): %s\n", result1)
	fmt.Printf("Expected: 1211e59e5e514882a6d44e684117b4c8\n\n")

	// 尝试使用硬编码的盐值 "ff"
	hardcodedSalt := "ff"
	hash2 := md5.Sum([]byte(hardcodedSalt + plaintext))
	result2 := hex.EncodeToString(hash2[:])
	fmt.Printf("Using hardcoded salt 'ff':\n")
	fmt.Printf("md5($salt.$pass): %s\n", result2)
	fmt.Printf("Expected: 1211e59e5e514882a6d44e684117b4c8\n")
}
