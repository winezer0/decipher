package main

import (
	"crypto/md5"
	"encoding/hex"
	"fmt"
)

func main() {
	// 用户提供的示例
	plaintext := "123456"
	secret := "66666"
	expected := "1211e59e5e514882a6d44e684117b4c8"

	fmt.Printf("Plaintext: %s\n", plaintext)
	fmt.Printf("Secret: %s\n", secret)
	fmt.Printf("Expected: %s\n\n", expected)

	// 尝试各种可能的组合

	// 组合1: salt + pass
	test1 := secret + plaintext
	hash1 := md5.Sum([]byte(test1))
	result1 := hex.EncodeToString(hash1[:])
	fmt.Printf("salt + pass: %s\n", test1)
	fmt.Printf("md5: %s", result1)
	if result1 == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// 组合2: pass + salt
	test2 := plaintext + secret
	hash2 := md5.Sum([]byte(test2))
	result2 := hex.EncodeToString(hash2[:])
	fmt.Printf("pass + salt: %s\n", test2)
	fmt.Printf("md5: %s", result2)
	if result2 == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// 组合3: salt的hex编码 + pass
	saltHex := hex.EncodeToString([]byte(secret))
	test3 := saltHex + plaintext
	hash3 := md5.Sum([]byte(test3))
	result3 := hex.EncodeToString(hash3[:])
	fmt.Printf("hex(salt) + pass: %s\n", test3)
	fmt.Printf("md5: %s", result3)
	if result3 == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// 组合4: pass + salt的hex编码
	test4 := plaintext + saltHex
	hash4 := md5.Sum([]byte(test4))
	result4 := hex.EncodeToString(hash4[:])
	fmt.Printf("pass + hex(salt): %s\n", test4)
	fmt.Printf("md5: %s", result4)
	if result4 == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// 组合5: 对salt进行hex解码（添加0处理奇数长度）
	saltHexPadded := secret
	if len(saltHexPadded)%2 != 0 {
		saltHexPadded = "0" + saltHexPadded
	}
	decodedSalt, err := hex.DecodeString(saltHexPadded)
	if err == nil {
		test5 := string(decodedSalt) + plaintext
		hash5 := md5.Sum([]byte(test5))
		result5 := hex.EncodeToString(hash5[:])
		fmt.Printf("hex_decode('0'+salt) + pass: %s\n", test5)
		fmt.Printf("md5: %s", result5)
		if result5 == expected {
			fmt.Println(" ✓")
		} else {
			fmt.Println()
		}
	}
}
