package main

import (
	"crypto/md5"
	"encoding/hex"
	"fmt"
	"io"
)

func md5Hex(s string) string {
	h := md5.New()
	io.WriteString(h, s)
	return hex.EncodeToString(h.Sum(nil))
}

func main() {
	// 计算空字符串的 md5(md5(""))
	h1 := md5Hex("")
	fmt.Printf("md5(\"\") = %s\n", h1)
	h2 := md5Hex(h1)
	fmt.Printf("md5(md5(\"\") = %s\n", h2)

	// 计算空字符串的 md5(md5($salt).$pass) 其中 salt 为空
	h3 := md5Hex("")
	h4 := md5Hex(h3 + "")
	fmt.Printf("md5(md5(\"\").\"\") = %s\n", h4)

	// 计算空字符串的 md5($pass.md5($salt)) 其中 salt 为空
	h5 := md5Hex("")
	h6 := md5Hex("" + h5)
	fmt.Printf("md5(\"\".md5(\"\")) = %s\n", h6)
}
