package main

import (
	"crypto/md5"
	"crypto/sha1"
	"encoding/hex"
	"fmt"
)

func main() {
	// 用户提供的示例
	plaintext := "123456"
	secret := "66666"

	fmt.Printf("Plaintext: %s\n", plaintext)
	fmt.Printf("Secret: %s\n\n", secret)

	// 测试 md5($salt.$pass)
	expected := "1211e59e5e514882a6d44e684117b4c8"
	hash := md5.Sum([]byte(secret + plaintext))
	result := hex.EncodeToString(hash[:])
	fmt.Printf("md5($salt.$pass): %s", result)
	if result == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
		fmt.Printf("Expected: %s\n", expected)
	}

	// 测试 md5($pass.$salt)
	expected2 := "28abf184c96d99f171a22ec666af5111"
	hash2 := md5.Sum([]byte(plaintext + secret))
	result2 := hex.EncodeToString(hash2[:])
	fmt.Printf("md5($pass.$salt): %s", result2)
	if result2 == expected2 {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
		fmt.Printf("Expected: %s\n", expected2)
	}

	// 测试 sha1($pass.$salt)
	expected3 := "12fe077c7ee4a90418863ae9b03399b32cc38e92"
	hash3 := sha1.Sum([]byte(plaintext + secret))
	result3 := hex.EncodeToString(hash3[:])
	fmt.Printf("sha1($pass.$salt): %s", result3)
	if result3 == expected3 {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
		fmt.Printf("Expected: %s\n", expected3)
	}

	// 测试 sha1($salt.$pass)
	expected4 := "b25c22c5e91f9fe95761e4a61f68b34353387b96"
	hash4 := sha1.Sum([]byte(secret + plaintext))
	result4 := hex.EncodeToString(hash4[:])
	fmt.Printf("sha1($salt.$pass): %s", result4)
	if result4 == expected4 {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
		fmt.Printf("Expected: %s\n", expected4)
	}
}
