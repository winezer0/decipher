package main

import (
	"crypto/md5"
	"encoding/hex"
	"fmt"
)

func main() {
	// 用户提供的示例
	plaintext := "123456"
	secret := "66666" // 假设这是一个hex字符串
	expected := "1211e59e5e514882a6d44e684117b4c8"

	fmt.Printf("Plaintext: %s\n", plaintext)
	fmt.Printf("Secret: %s\n", secret)
	fmt.Printf("Expected: %s\n\n", expected)

	// 对salt进行hex解码（添加0处理奇数长度）
	saltHex := secret
	if len(saltHex)%2 != 0 {
		saltHex = "0" + saltHex
	}
	decodedSalt, err := hex.DecodeString(saltHex)
	if err != nil {
		fmt.Printf("Error decoding salt: %v\n", err)
		return
	}
	fmt.Printf("Salt hex: %s\n", saltHex)
	fmt.Printf("Decoded salt: %v\n", decodedSalt)
	fmt.Printf("Decoded salt (string): %s\n\n", string(decodedSalt))

	// 测试 md5($salt.$pass)
	combined := string(decodedSalt) + plaintext
	hash := md5.Sum([]byte(combined))
	result := hex.EncodeToString(hash[:])
	fmt.Printf("Combined: %v\n", combined)
	fmt.Printf("md5($salt.$pass): %s", result)
	if result == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// 测试其他算法
	// md5($pass.$salt)
	expected2 := "28abf184c96d99f171a22ec666af5111"
	combined2 := plaintext + string(decodedSalt)
	hash2 := md5.Sum([]byte(combined2))
	result2 := hex.EncodeToString(hash2[:])
	fmt.Printf("md5($pass.$salt): %s", result2)
	if result2 == expected2 {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}
}
