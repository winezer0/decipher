package main

import (
	"crypto/md5"
	"encoding/hex"
	"fmt"
)

func main() {
	plaintext := "123456"
	secret := "66666"

	// 方法：组合盐和明文，然后对整个组合进行HEX编码
	combined := secret + plaintext
	encodedCombined := hex.EncodeToString([]byte(combined))
	hash := md5.Sum([]byte(encodedCombined))
	fmt.Printf("组合后HEX编码: %x\n", hash)

	// 预期结果
	fmt.Printf("预期结果: 1211e59e5e514882a6d44e684117b4c8\n")
}
