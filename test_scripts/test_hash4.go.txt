package main

import (
	"crypto/md5"
	"encoding/hex"
	"fmt"
)

func main() {
	plaintext := "123456"
	secret := "66666" // 假设这是一个hex字符串

	// 测试预期值
	expected := "1211e59e5e514882a6d44e684117b4c8"
	fmt.Printf("Expected: %s\n\n", expected)

	// 尝试解码salt（添加0来处理奇数长度）
	saltHex := secret
	if len(saltHex)%2 != 0 {
		saltHex = "0" + saltHex
	}
	decodedSalt, err := hex.DecodeString(saltHex)
	if err != nil {
		fmt.Printf("Error decoding salt: %v\n", err)
		return
	}
	fmt.Printf("Decoded salt: %s\n\n", string(decodedSalt))

	// 测试各种可能的组合

	// Test 1: md5($pass.$salt)
	hash := md5.Sum([]byte(plaintext + string(decodedSalt)))
	result := hex.EncodeToString(hash[:])
	fmt.Printf("md5($pass.$salt): %s", result)
	if result == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// Test 2: md5($salt.$pass)
	hash = md5.Sum([]byte(string(decodedSalt) + plaintext))
	result = hex.EncodeToString(hash[:])
	fmt.Printf("md5($salt.$pass): %s", result)
	if result == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// Test 3: md5(md5($pass).md5($salt))
	md5Pass := md5.Sum([]byte(plaintext))
	md5Salt := md5.Sum(decodedSalt)
	hash = md5.Sum([]byte(hex.EncodeToString(md5Pass[:]) + hex.EncodeToString(md5Salt[:])))
	result = hex.EncodeToString(hash[:])
	fmt.Printf("md5(md5($pass).md5($salt)): %s", result)
	if result == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}
}
