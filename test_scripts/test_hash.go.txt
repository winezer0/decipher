package main

import (
	"crypto/md5"
	"crypto/sha1"
	"crypto/sha256"
	"crypto/sha512"
	"encoding/hex"
	"fmt"
)

func main() {
	plaintext := "123456"
	secret := "66666"

	// Test md5($pass.$salt)
	hash := md5.Sum([]byte(plaintext + secret))
	fmt.Printf("md5($pass.$salt): %s\n", hex.EncodeToString(hash[:]))

	// Test md5($salt.$pass)
	hash = md5.Sum([]byte(secret + plaintext))
	fmt.Printf("md5($salt.$pass): %s\n", hex.EncodeToString(hash[:]))

	// Test md5(md5($pass).md5($salt))
	md5Pass := md5.Sum([]byte(plaintext))
	md5Salt := md5.Sum([]byte(secret))
	hash = md5.Sum([]byte(hex.EncodeToString(md5Pass[:]) + hex.EncodeToString(md5Salt[:])))
	fmt.Printf("md5(md5($pass).md5($salt)): %s\n", hex.EncodeToString(hash[:]))

	// Test md5(md5($salt).md5($pass))
	hash = md5.Sum([]byte(hex.EncodeToString(md5Salt[:]) + hex.EncodeToString(md5Pass[:])))
	fmt.Printf("md5(md5($salt).md5($pass)): %s\n", hex.EncodeToString(hash[:]))

	// Test md5($salt.$pass.$salt)
	hash = md5.Sum([]byte(secret + plaintext + secret))
	fmt.Printf("md5($salt.$pass.$salt): %s\n", hex.EncodeToString(hash[:]))

	// Test md5(md5($salt).$pass)
	hash = md5.Sum([]byte(hex.EncodeToString(md5Salt[:]) + plaintext))
	fmt.Printf("md5(md5($salt).$pass): %s\n", hex.EncodeToString(hash[:]))

	// Test md5($salt.md5($pass))
	hash = md5.Sum([]byte(secret + hex.EncodeToString(md5Pass[:])))
	fmt.Printf("md5($salt.md5($pass)): %s\n", hex.EncodeToString(hash[:]))

	// Test md5(md5($pass).$salt)
	hash = md5.Sum([]byte(hex.EncodeToString(md5Pass[:]) + secret))
	fmt.Printf("md5(md5($pass).$salt): %s\n", hex.EncodeToString(hash[:]))

	// Test md5($pass.md5($salt))
	hash = md5.Sum([]byte(plaintext + hex.EncodeToString(md5Salt[:])))
	fmt.Printf("md5($pass.md5($salt)): %s\n", hex.EncodeToString(hash[:]))

	// Test sha1($pass.$salt)
	hash1 := sha1.Sum([]byte(plaintext + secret))
	fmt.Printf("sha1($pass.$salt): %s\n", hex.EncodeToString(hash1[:]))

	// Test sha1($salt.$pass)
	hash1 = sha1.Sum([]byte(secret + plaintext))
	fmt.Printf("sha1($salt.$pass): %s\n", hex.EncodeToString(hash1[:]))

	// Test sha256($salt.$pass)
	hash256 := sha256.Sum256([]byte(secret + plaintext))
	fmt.Printf("sha256($salt.$pass): %s\n", hex.EncodeToString(hash256[:]))

	// Test sha512($pass.$salt)
	hash512 := sha512.Sum512([]byte(plaintext + secret))
	fmt.Printf("sha512($pass.$salt): %s\n", hex.EncodeToString(hash512[:]))

	// Test sha256($pass.$salt)
	hash256 = sha256.Sum256([]byte(plaintext + secret))
	fmt.Printf("sha256($pass.$salt): %s\n", hex.EncodeToString(hash256[:]))

	// Test sha512($salt.$pass)
	hash512 = sha512.Sum512([]byte(secret + plaintext))
	fmt.Printf("sha512($salt.$pass): %s\n", hex.EncodeToString(hash512[:]))
}
