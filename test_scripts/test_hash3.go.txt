package main

import (
	"crypto/md5"
	"encoding/hex"
	"fmt"
)

func main() {
	plaintext := "123456"
	secret := "66666"

	// 测试预期值
	expected := "1211e59e5e514882a6d44e684117b4c8"
	fmt.Printf("Expected: %s\n\n", expected)

	// 测试各种可能的组合

	// Test 1: md5($pass.$salt)
	hash := md5.Sum([]byte(plaintext + secret))
	result := hex.EncodeToString(hash[:])
	fmt.Printf("md5($pass.$salt): %s", result)
	if result == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// Test 2: md5($salt.$pass)
	hash = md5.Sum([]byte(secret + plaintext))
	result = hex.EncodeToString(hash[:])
	fmt.Printf("md5($salt.$pass): %s", result)
	if result == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// Test 3: md5(hex($pass).hex($salt))
	hexPass := hex.EncodeToString([]byte(plaintext))
	hexSalt := hex.EncodeToString([]byte(secret))
	hash = md5.Sum([]byte(hexPass + hexSalt))
	result = hex.EncodeToString(hash[:])
	fmt.Printf("md5(hex($pass).hex($salt)): %s", result)
	if result == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// Test 4: md5(hex($salt).hex($pass))
	hash = md5.Sum([]byte(hexSalt + hexPass))
	result = hex.EncodeToString(hash[:])
	fmt.Printf("md5(hex($salt).hex($pass)): %s", result)
	if result == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// Test 5: md5(md5($pass).$salt)
	md5Pass := md5.Sum([]byte(plaintext))
	hash = md5.Sum([]byte(hex.EncodeToString(md5Pass[:]) + secret))
	result = hex.EncodeToString(hash[:])
	fmt.Printf("md5(md5($pass).$salt): %s", result)
	if result == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// Test 6: md5($pass.md5($salt))
	md5Salt := md5.Sum([]byte(secret))
	hash = md5.Sum([]byte(plaintext + hex.EncodeToString(md5Salt[:])))
	result = hex.EncodeToString(hash[:])
	fmt.Printf("md5($pass.md5($salt)): %s", result)
	if result == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// Test 7: md5(md5($salt).$pass)
	hash = md5.Sum([]byte(hex.EncodeToString(md5Salt[:]) + plaintext))
	result = hex.EncodeToString(hash[:])
	fmt.Printf("md5(md5($salt).$pass): %s", result)
	if result == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// Test 8: md5($salt.md5($pass))
	hash = md5.Sum([]byte(secret + hex.EncodeToString(md5Pass[:])))
	result = hex.EncodeToString(hash[:])
	fmt.Printf("md5($salt.md5($pass)): %s", result)
	if result == expected {
		fmt.Println(" ✓")
	} else {
		fmt.Println()
	}

	// Test 9: md5(hex_decode($salt).$pass) - with padding
	decodedSalt, err := hex.DecodeString("0" + secret)
	if err == nil {
		hash = md5.Sum([]byte(string(decodedSalt) + plaintext))
		result = hex.EncodeToString(hash[:])
		fmt.Printf("md5(hex_decode('0'+$salt).$pass): %s", result)
		if result == expected {
			fmt.Println(" ✓")
		} else {
			fmt.Println()
		}
	}
}
