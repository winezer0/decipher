package main

import (
	"crypto/md5"
	"encoding/hex"
	"fmt"
)

func main() {
	plaintext := "123456"
	secret := "66666"

	// Test 1: 直接使用盐
	hash := md5.Sum([]byte(secret + plaintext))
	fmt.Printf("md5($salt.$pass): %s\n", hex.EncodeToString(hash[:]))

	// Test 2: 对盐进行hex编码
	hexSalt := hex.EncodeToString([]byte(secret))
	hash = md5.Sum([]byte(hexSalt + plaintext))
	fmt.Printf("md5(hex($salt).$pass): %s\n", hex.EncodeToString(hash[:]))

	// Test 3: 对盐进行hex解码（如果盐是hex字符串）
	// 注意："66666" 不是有效的hex字符串，因为长度是奇数
	// 尝试解码 "66666"（长度为5，奇数）
	decodedSalt, err := hex.DecodeString(secret)
	if err != nil {
		fmt.Printf("Error decoding hex: %v\n", err)
		// 尝试在前面加0
		decodedSalt, err = hex.DecodeString("0" + secret)
		if err != nil {
			fmt.Printf("Error decoding hex with padding: %v\n", err)
		} else {
			hash = md5.Sum([]byte(string(decodedSalt) + plaintext))
			fmt.Printf("md5(hex_decode('0'+$salt).$pass): %s\n", hex.EncodeToString(hash[:]))
		}
	} else {
		hash = md5.Sum([]byte(string(decodedSalt) + plaintext))
		fmt.Printf("md5(hex_decode($salt).$pass): %s\n", hex.EncodeToString(hash[:]))
	}

	// 测试预期值
	expected := "1211e59e5e514882a6d44e684117b4c8"
	fmt.Printf("Expected: %s\n", expected)
}
