package main

import (
	"crypto/md5"
	"encoding/hex"
	"fmt"
)

func main() {
	plaintext := "123456"
	secret := "66666"

	// 方法：先计算普通的MD5哈希，然后对结果进行HEX编码
	hash := md5.Sum([]byte(secret + plaintext))
	hexHash := hex.EncodeToString(hash[:])
	fmt.Printf("哈希后HEX编码: %s\n", hexHash)

	// 预期结果
	fmt.Printf("预期结果: 1211e59e5e514882a6d44e684117b4c8\n")
}
