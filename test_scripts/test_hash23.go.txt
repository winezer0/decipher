package main

import (
	"crypto/md5"
	"encoding/hex"
	"fmt"
)

func main() {
	plaintext := "123456"
	secret := "66666"

	// 方法：对盐进行HEX解码，使用不同的方式
	// 方式1：在前面添加0
	decoded1, err1 := hex.DecodeString("0" + secret)
	if err1 == nil {
		hash1 := md5.Sum([]byte(string(decoded1) + plaintext))
		fmt.Printf("方式1 (前导0): %x\n", hash1)
	}

	// 方式2：在后面添加0
	decoded2, err2 := hex.DecodeString(secret + "0")
	if err2 == nil {
		hash2 := md5.Sum([]byte(string(decoded2) + plaintext))
		fmt.Printf("方式2 (后导0): %x\n", hash2)
	}

	// 方式3：只取前4个字符
	if len(secret) >= 4 {
		decoded3, err3 := hex.DecodeString(secret[:4])
		if err3 == nil {
			hash3 := md5.Sum([]byte(string(decoded3) + plaintext))
			fmt.Printf("方式3 (前4字符): %x\n", hash3)
		}
	}

	// 方式4：只取后4个字符
	if len(secret) >= 4 {
		decoded4, err4 := hex.DecodeString(secret[len(secret)-4:])
		if err4 == nil {
			hash4 := md5.Sum([]byte(string(decoded4) + plaintext))
			fmt.Printf("方式4 (后4字符): %x\n", hash4)
		}
	}

	// 预期结果
	fmt.Printf("预期结果: 1211e59e5e514882a6d44e684117b4c8\n")
}
