package main

import (
	"crypto/md5"
	"crypto/sha1"
	"crypto/sha256"
	"crypto/sha512"
	"encoding/hex"
	"fmt"
)

// DecodeHexSafe 安全地解码 hex 字符串，处理奇数长度的情况
func DecodeHexSafe(s string) ([]byte, error) {
	if len(s)%2 != 0 {
		s = "0" + s
	}
	return hex.DecodeString(s)
}

func main() {
	// 用户提供的示例
	plaintext := "123456"
	secret := "66666"

	// 对salt进行hex解码，处理奇数长度的情况
	decodedSalt, err := DecodeHexSafe(secret)
	if err != nil {
		fmt.Printf("Error decoding salt: %v\n", err)
		return
	}

	fmt.Printf("Plaintext: %s\n", plaintext)
	fmt.Printf("Secret: %s\n", secret)
	fmt.Printf("Decoded salt: %s\n\n", string(decodedSalt))

	// 计算各种组合的哈希值

	// md5($salt.$pass)
	hash1 := md5.Sum([]byte(string(decodedSalt) + plaintext))
	result1 := hex.EncodeToString(hash1[:])
	fmt.Printf("md5($salt.$pass): %s\n", result1)

	// md5($pass.$salt)
	hash2 := md5.Sum([]byte(plaintext + string(decodedSalt)))
	result2 := hex.EncodeToString(hash2[:])
	fmt.Printf("md5($pass.$salt): %s\n", result2)

	// md5(md5($pass).md5($salt))
	md5Pass := md5.Sum([]byte(plaintext))
	md5Salt := md5.Sum([]byte(string(decodedSalt)))
	hash3 := md5.Sum([]byte(hex.EncodeToString(md5Pass[:]) + hex.EncodeToString(md5Salt[:])))
	result3 := hex.EncodeToString(hash3[:])
	fmt.Printf("md5(md5($pass).md5($salt)): %s\n", result3)

	// md5(md5($salt).md5($pass))
	hash4 := md5.Sum([]byte(hex.EncodeToString(md5Salt[:]) + hex.EncodeToString(md5Pass[:])))
	result4 := hex.EncodeToString(hash4[:])
	fmt.Printf("md5(md5($salt).md5($pass)): %s\n", result4)

	// md5($salt.$pass.$salt)
	hash5 := md5.Sum([]byte(string(decodedSalt) + plaintext + string(decodedSalt)))
	result5 := hex.EncodeToString(hash5[:])
	fmt.Printf("md5($salt.$pass.$salt): %s\n", result5)

	// md5(md5($salt).$pass)
	hash6 := md5.Sum([]byte(hex.EncodeToString(md5Salt[:]) + plaintext))
	result6 := hex.EncodeToString(hash6[:])
	fmt.Printf("md5(md5($salt).$pass): %s\n", result6)

	// md5($salt.md5($pass))
	hash7 := md5.Sum([]byte(string(decodedSalt) + hex.EncodeToString(md5Pass[:])))
	result7 := hex.EncodeToString(hash7[:])
	fmt.Printf("md5($salt.md5($pass)): %s\n", result7)

	// md5(md5($pass).$salt)
	hash8 := md5.Sum([]byte(hex.EncodeToString(md5Pass[:]) + string(decodedSalt)))
	result8 := hex.EncodeToString(hash8[:])
	fmt.Printf("md5(md5($pass).$salt): %s\n", result8)

	// md5($pass.md5($salt))
	hash9 := md5.Sum([]byte(plaintext + hex.EncodeToString(md5Salt[:])))
	result9 := hex.EncodeToString(hash9[:])
	fmt.Printf("md5($pass.md5($salt)): %s\n", result9)

	// SHA1
	sha1PassSalt := sha1.Sum([]byte(plaintext + string(decodedSalt)))
	resultSHA1PassSalt := hex.EncodeToString(sha1PassSalt[:])
	fmt.Printf("sha1($pass.$salt): %s\n", resultSHA1PassSalt)

	sha1SaltPass := sha1.Sum([]byte(string(decodedSalt) + plaintext))
	resultSHA1SaltPass := hex.EncodeToString(sha1SaltPass[:])
	fmt.Printf("sha1($salt.$pass): %s\n", resultSHA1SaltPass)

	// SHA256
	sha256PassSalt := sha256.Sum256([]byte(plaintext + string(decodedSalt)))
	resultSHA256PassSalt := hex.EncodeToString(sha256PassSalt[:])
	fmt.Printf("sha256($pass.$salt): %s\n", resultSHA256PassSalt)

	sha256SaltPass := sha256.Sum256([]byte(string(decodedSalt) + plaintext))
	resultSHA256SaltPass := hex.EncodeToString(sha256SaltPass[:])
	fmt.Printf("sha256($salt.$pass): %s\n", resultSHA256SaltPass)

	// SHA512
	sha512PassSalt := sha512.Sum512([]byte(plaintext + string(decodedSalt)))
	resultSHA512PassSalt := hex.EncodeToString(sha512PassSalt[:])
	fmt.Printf("sha512($pass.$salt): %s\n", resultSHA512PassSalt)

	sha512SaltPass := sha512.Sum512([]byte(string(decodedSalt) + plaintext))
	resultSHA512SaltPass := hex.EncodeToString(sha512SaltPass[:])
	fmt.Printf("sha512($salt.$pass): %s\n", resultSHA512SaltPass)

	// 输出用户提供的预期值
	fmt.Println("\nUser provided expected values:")
	fmt.Println("md5(md5($pass).md5($salt)): 087cc69db39298e93fcc0a4e6e649847:66666")
	fmt.Println("md5($salt.$pass): 1211e59e5e514882a6d44e684117b4c8:66666")
	fmt.Println("md5(md5($salt).md5($pass)): c2058a78cb695dfcc1366aeb854c88ba:66666")
	fmt.Println("md5($salt.$pass.$salt): f32b60d3a5fcb26ea73e2a7ac3bc37af:66666")
	fmt.Println("md5($pass.$salt): 28abf184c96d99f171a22ec666af5111:66666")
	fmt.Println("md5(md5($salt).$pass): 16be1e15122733e95ab98d50e742f23b:66666")
	fmt.Println("md5($salt.md5($pass)): 59390fc41c51081308aa5663a4825519:66666")
	fmt.Println("md5(md5($pass).$salt);VB;DZ: c3c9c11d89af8bb620a7bd08cccfc3ca:66666")
	fmt.Println("md5($pass.md5($salt)): 046ec44e4e8cc8cb25198ab13a50e2b8:66666")
	fmt.Println("sha1($pass.$salt): 12fe077c7ee4a90418863ae9b03399b32cc38e92:66666")
	fmt.Println("sha1($salt.$pass): b25c22c5e91f9fe95761e4a61f68b34353387b96:66666")
	fmt.Println("sha256($salt.$pass): d771132aef8ea03da584b940a9fe3c27a98a9e5724675817094b1f67500182d6:66666")
	fmt.Println("sha512($pass.$salt): 385411ebd18e63325a78f999e6a8ed6b4e4c09bfcc626e23b8df9a068edd90a7450131fbc91eededffe05200d61f674b08b95718e447c92bd2d8fceaf80554ca:66666")
	fmt.Println("sha256($pass.$salt): d0eec2347309dc687be910836167ae0b87aae89caa03963604690ff3200a191b:66666")
	fmt.Println("sha512($salt.$pass): 1fefddb440e405e08c93e36cb20fabb3f304aaed8fc959f149ac5e007293d29c159b67037e924794b1b1413a4f772172e8c1d84461a271f6d91385bc3353a87c:66666")
}
