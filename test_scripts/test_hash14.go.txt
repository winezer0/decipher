package main

import (
	"crypto/md5"
	"crypto/sha1"
	"crypto/sha256"
	"crypto/sha512"
	"encoding/hex"
	"fmt"
)

func main() {
	// 用户提供的示例
	plaintext := "123456"
	secret := "66666"

	fmt.Printf("Plaintext: %s\n", plaintext)
	fmt.Printf("Secret: %s\n\n", secret)

	// 计算各种组合的哈希值

	// md5($salt.$pass)
	hash1 := md5.Sum([]byte(secret + plaintext))
	result1 := hex.EncodeToString(hash1[:])
	fmt.Printf("md5($salt.$pass): %s\n", result1)
	fmt.Printf("Expected: 1211e59e5e514882a6d44e684117b4c8\n\n")

	// md5($pass.$salt)
	hash2 := md5.Sum([]byte(plaintext + secret))
	result2 := hex.EncodeToString(hash2[:])
	fmt.Printf("md5($pass.$salt): %s\n", result2)
	fmt.Printf("Expected: 28abf184c96d99f171a22ec666af5111\n\n")

	// md5($salt.$pass.$salt)
	hash5 := md5.Sum([]byte(secret + plaintext + secret))
	result5 := hex.EncodeToString(hash5[:])
	fmt.Printf("md5($salt.$pass.$salt): %s\n", result5)
	fmt.Printf("Expected: f32b60d3a5fcb26ea73e2a7ac3bc37af\n\n")

	// 尝试其他可能的组合
	fmt.Println("Trying other combinations:")

	// 尝试使用不同的编码方式
	saltBytes := []byte(secret)
	fmt.Printf("Salt bytes: %v\n", saltBytes)

	// 尝试对整个组合进行 hex 编码
	combined := secret + plaintext
	combinedHex := hex.EncodeToString([]byte(combined))
	hashCombined := md5.Sum([]byte(combinedHex))
	resultCombined := hex.EncodeToString(hashCombined[:])
	fmt.Printf("md5(hex($salt.$pass)): %s\n", resultCombined)

	// 尝试对明文进行 hex 编码
	plaintextHex := hex.EncodeToString([]byte(plaintext))
	hashPlaintextHex := md5.Sum([]byte(secret + plaintextHex))
	resultPlaintextHex := hex.EncodeToString(hashPlaintextHex[:])
	fmt.Printf("md5($salt.hex($pass)): %s\n", resultPlaintextHex)

	// 尝试对盐和明文都进行 hex 编码
	saltHex := hex.EncodeToString(saltBytes)
	plaintextHex = hex.EncodeToString([]byte(plaintext))
	hashBothHex := md5.Sum([]byte(saltHex + plaintextHex))
	resultBothHex := hex.EncodeToString(hashBothHex[:])
	fmt.Printf("md5(hex($salt).hex($pass)): %s\n", resultBothHex)

	// 尝试 SHA1
	sha1PassSalt := sha1.Sum([]byte(plaintext + secret))
	resultSHA1PassSalt := hex.EncodeToString(sha1PassSalt[:])
	fmt.Printf("\nsha1($pass.$salt): %s\n", resultSHA1PassSalt)
	fmt.Printf("Expected: 12fe077c7ee4a90418863ae9b03399b32cc38e92\n")

	// 尝试 SHA256
	sha256PassSalt := sha256.Sum256([]byte(plaintext + secret))
	resultSHA256PassSalt := hex.EncodeToString(sha256PassSalt[:])
	fmt.Printf("\nsha256($pass.$salt): %s\n", resultSHA256PassSalt)
	fmt.Printf("Expected: d0eec2347309dc687be910836167ae0b87aae89caa03963604690ff3200a191b\n")

	// 尝试 SHA512
	sha512PassSalt := sha512.Sum512([]byte(plaintext + secret))
	resultSHA512PassSalt := hex.EncodeToString(sha512PassSalt[:])
	fmt.Printf("\nsha512($pass.$salt): %s\n", resultSHA512PassSalt)
	fmt.Printf("Expected: 385411ebd18e63325a78f999e6a8ed6b4e4c09bfcc626e23b8df9a068edd90a7450131fbc91eededffe05200d61f674b08b95718e447c92bd2d8fceaf80554ca\n")
}
