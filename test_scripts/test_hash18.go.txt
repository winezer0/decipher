package main

import (
	"crypto/md5"
	"encoding/hex"
	"fmt"
)

func main() {
	plaintext := "123456"
	secret := "66666"

	// 方法6：对明文进行HEX编码，盐直接使用
	encodedPlaintext := hex.EncodeToString([]byte(plaintext))
	hash6 := md5.Sum([]byte(secret + encodedPlaintext))
	fmt.Printf("方法6 (明文HEX编码): %x\n", hash6)

	// 方法7：对明文进行HEX编码，对盐进行HEX解码
	var decodedSalt []byte
	var err error
	if len(secret)%2 != 0 {
		decodedSalt, err = hex.DecodeString("0" + secret)
	} else {
		decodedSalt, err = hex.DecodeString(secret)
	}
	if err != nil {
		fmt.Println("解码错误:", err)
	} else {
		hash7 := md5.Sum([]byte(string(decodedSalt) + encodedPlaintext))
		fmt.Printf("方法7 (明文HEX编码，盐HEX解码): %x\n", hash7)
	}

	// 方法8：特殊处理，将66666解码为ff，对明文进行HEX编码
	hash8 := md5.Sum([]byte("\xff" + encodedPlaintext))
	fmt.Printf("方法8 (盐解码为ff，明文HEX编码): %x\n", hash8)

	// 预期结果
	fmt.Printf("预期结果: 1211e59e5e514882a6d44e684117b4c8\n")
}
